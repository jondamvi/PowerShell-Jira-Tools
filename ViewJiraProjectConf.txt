param(
    [Parameter(Mandatory)][string]$BaseUrl,   # https://yoursite.atlassian.net
    [Parameter(Mandatory)][string]$Email,
    [Parameter(Mandatory)][string]$Token
)
$h = @{
    Authorization  = "Basic " + [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes("${Email}:${Token}"))
    "Content-Type" = "application/json"
}
function Get-J($url) {
    try { Invoke-RestMethod -Uri $url -Headers $h -ErrorAction Stop }
    catch {
        $resp = $_.Exception.Response
        if ($resp -and $resp.StatusCode.value__ -in 401,403) { "DENIED" } else { $null }
    }
}
$projs = @(); $start = 0
do {
    $r = Get-J "$BaseUrl/rest/api/3/project/search?maxResults=50&startAt=$start&expand=lead"
    if ($r -eq "DENIED") { Write-Warning "Cannot list projects: access denied"; exit }
    if (!$r) { break }
    if ($r.values) { $projs += $r.values }
    $start += 50
} while (-not $r.isLast)

$uCache = @{}; $psCache = @{}; $gCache = @{}
function Get-US($id) {
    if (!$uCache.ContainsKey($id)) {
        $u = Get-J "$BaseUrl/rest/api/3/user?accountId=$id"
        $uCache[$id] = if ($u -eq "DENIED") {"DENIED"} elseif (!$u) {"MISSING"} elseif (!$u.active) {"INACTIVE"} else {"OK"}
    }
    $uCache[$id]
}
function Get-GS($id) {
    if (!$gCache.ContainsKey($id)) {
        $g = Get-J "$BaseUrl/rest/api/3/group?groupId=$id"
        $gCache[$id] = if ($g) {"OK"} else {"MISSING"}
    }
    $gCache[$id]
}

$out = foreach ($p in $projs) {
    $iss = @()
    if (!$p.lead) { $iss += "No project lead" }
    else {
        $us = Get-US $p.lead.accountId
        if ($us -ne "OK") { $iss += "Project lead: $($us.ToLower())" }
    }
    $ps = Get-J "$BaseUrl/rest/api/3/project/$($p.key)/permissionscheme"
    if     ($ps -eq "DENIED") { $iss += "PERMISSION DENIED: scheme" }
    elseif (!$ps)              { $iss += "No permission scheme assigned" }
    else {
        if (!$psCache.ContainsKey($ps.id)) {
            $psCache[$ps.id] = Get-J "$BaseUrl/rest/api/3/permissionscheme/$($ps.id)?expand=permissions"
        }
        $sd = $psCache[$ps.id]
        if ($sd -and $sd -ne "DENIED") {
            if (!($sd.permissions | Where-Object { $_.permission -eq "BROWSE_PROJECTS" })) {
                $iss += "Scheme: BROWSE_PROJECTS not granted"
            }
            foreach ($perm in $sd.permissions) {
                if ($perm.holder.type -eq "user") {
                    $us = Get-US $perm.holder.parameter
                    if ($us -in "MISSING","INACTIVE") { $iss += "Scheme: $($us.ToLower()) user in $($perm.permission)" }
                } elseif ($perm.holder.type -eq "group") {
                    if ((Get-GS $perm.holder.parameter) -eq "MISSING") {
                        $iss += "Scheme: group '$($perm.holder.value)' missing"
                    }
                }
            }
        }
    }
    $comps = Get-J "$BaseUrl/rest/api/3/project/$($p.key)/components"
    if ($comps -and $comps -ne "DENIED") {
        foreach ($comp in ($comps | Where-Object { $_.lead })) {
            $us = Get-US $comp.lead.accountId
            if ($us -in "MISSING","INACTIVE") { $iss += "Component '$($comp.name)': lead $($us.ToLower())" }
        }
    }
    if ($iss.Count) { [PSCustomObject]@{ Name=$p.name; Key=$p.key; Id=$p.id; Issues=($iss -join " | ") } }
}
if ($out) { $out | Format-Table -AutoSize -Wrap } else { Write-Host "No configuration issues found." }
