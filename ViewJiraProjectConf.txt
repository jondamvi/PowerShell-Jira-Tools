    $ps = Get-J "$BaseUrl/rest/api/3/project/$($p.key)/permissionscheme"
    if     ($ps -eq "DENIED") { $iss += "DENIED: scheme" }
    elseif (!$ps)              { $iss += "No permission scheme" }
    else {
        if (!$psCache.ContainsKey($ps.id)) {
            $psCache[$ps.id] = Get-J "$BaseUrl/rest/api/3/permissionscheme/$($ps.id)?expand=permissions"
        }
        $sd = $psCache[$ps.id]
        if ($sd -and $sd -ne "DENIED") {
            if (!($sd.permissions | Where-Object { $_.permission -eq "BROWSE_PROJECTS" })) { $iss += "BROWSE_PROJECTS missing" }
            $missingGroups = @{}
            foreach ($perm in $sd.permissions) {
                if ($perm.holder.type -eq "user") {
                    $us = Get-US $perm.holder.parameter
                    if ($us -in "MISSING","INACTIVE") { $iss += "Scheme: $us user in $($perm.permission)" }
                } elseif ($perm.holder.type -eq "group") {
                    $gid = $perm.holder.parameter
                    if (!$gCache.ContainsKey($gid)) {
                        $g = Get-J "$BaseUrl/rest/api/3/group?groupId=$gid"
                        $gCache[$gid] = if ($g) { $g.name } else { "MISSING" }
                    }
                    if ($gCache[$gid] -eq "MISSING" -and !$missingGroups.ContainsKey($gid)) {
                        $missingGroups[$gid] = if ($perm.holder.value) { $perm.holder.value } else { $gid }
                    }
                }
            }
            foreach ($kv in $missingGroups.GetEnumerator()) {
                $iss += "Scheme: group '$($kv.Value)' [$($kv.Key)] missing"
            }
        }
    }
